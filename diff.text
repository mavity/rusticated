+func wordBoundary(value []rune, cursor, direction int) int {
+	isWord := func(r rune) bool {
+		return r == '_' || r >= '0' && r <= '9' || r >= 'a' && r <= 'z' || r >= 'A' && r <= 'Z' || r > 127
+	}
+	if direction < 0 {
+		for cursor > 0 && !isWord(value[cursor-1]) {
+			cursor--
+		}
+		for cursor > 0 && isWord(value[cursor-1]) {
+			cursor--
+		}
+		return cursor
+	}
+	for cursor < len(value) && !isWord(value[cursor]) {
+		cursor++
+	}
+	for cursor < len(value) && isWord(value[cursor]) {
+		cursor++
+	}
+	return cursor
+}
